cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET "esp32p4")
set(COMPONENTS main)
if(NOT DEFINED PROJECT_VER)
    set(PROJECT_VER "0.6.5-backlight-100hz")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(klipper_jc4880p443_solid)
idf_build_get_property(orientation_idf_version IDF_VER)
if(NOT orientation_idf_version STREQUAL "v5.5.6")
    message(FATAL_ERROR "Este proyecto requiere ESP-IDF v5.5.6")
endif()
# Match the renderer optimisation accepted in the existing base with Wi-Fi.
idf_component_get_property(orientation_lvgl_lib lvgl__lvgl COMPONENT_LIB)
target_compile_options(${orientation_lvgl_lib} PRIVATE -O2)
idf_component_get_property(orientation_transform_lib display_orientation COMPONENT_LIB)
target_compile_options(${orientation_transform_lib} PRIVATE -O2)
if(CONFIG_ESP_HOSTED_ENABLED)
    # Preserve the accepted ordering: GUI first, explicit Hosted init in worker.
    idf_component_get_property(panel_hosted_lib espressif__esp_hosted COMPONENT_LIB)
    get_target_property(panel_hosted_sources ${panel_hosted_lib} SOURCES)
    list(LENGTH panel_hosted_sources panel_hosted_before)
    list(FILTER panel_hosted_sources EXCLUDE REGEX "port_esp_hosted_host_init\\.c$")
    list(LENGTH panel_hosted_sources panel_hosted_after)
    math(EXPR panel_hosted_removed "${panel_hosted_before} - ${panel_hosted_after}")
    if(NOT panel_hosted_removed EQUAL 1)
        message(FATAL_ERROR "Expected exactly one Hosted 2.12.13 constructor source")
    endif()
    set_property(TARGET ${panel_hosted_lib} PROPERTY SOURCES "${panel_hosted_sources}")
endif()
